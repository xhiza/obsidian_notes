#!/usr/bin/env python3

import json
import sys
import time
from pathlib import Path
from typing import Dict, Any, Optional, Tuple
import requests
import requests.auth

class CreateDeploymentClient:
    def __init__(self, config_file: str = "create_deployment.conf"):
        self.config_file = config_file
        self.config = self._load_config()
        self.c4_retries = int(self.config.get("C4_RETRIES", 3))
        self.c4_retry_sleep = int(self.config.get("C4_RETRY_SLEEP", 2))

    def _load_config(self) -> Dict[str, str]:
        """Load configuration from bash-style config file"""
        if not Path(self.config_file).exists():
            raise FileNotFoundError(f"Error: {self.config_file} file not found!")

        config = {}
        with open(self.config_file, 'r') as f:
            for line in f:
                line = line.strip()
                # Skip comments and empty lines
                if not line or line.startswith('#'):
                    continue

                # Parse bash-style variable assignments
                if '=' in line:
                    key, value = line.split('=', 1)
                    key = key.strip()
                    value = value.strip()

                    # Remove quotes if present
                    if value.startswith('"') and value.endswith('"'):
                        value = value[1:-1]
                    elif value.startswith("'") and value.endswith("'"):
                        value = value[1:-1]

                    config[key] = value

        return config

    def _validate_config(self) -> None:
        """Validate required configuration parameters"""
        # Basic required vars for all requests
        basic_vars = ["HOST_URL", "ROUTE", "USER", "PASSWORD"]

        # Get request type (default to job_exec)
        request_type = self.config.get("REQUEST_TYPE", "job_exec")

        if request_type == "create_deployment":
            required_vars = basic_vars + [
                "DB_NAME", "WORKING_COPY", "PLATFORM", "INSTANCE_TYPE",
                "NUM_NODES", "SIZE", "REGION", "ACCOUNT_UUID", "DB_UUID",
                "CLUSTER_UUID", "CLIENT_ID", "MAX_MULTI_CLUSTER_NODES", "MAX_CLUSTERS"
            ]
        elif request_type == "job_exec":
            required_vars = basic_vars + ["JOB_NAME"]
        else:
            print(f"Error: Invalid REQUEST_TYPE '{request_type}'. Must be 'create_deployment' or 'job_exec'")
            sys.exit(1)

        missing_vars = []
        for var in required_vars:
            if not self.config.get(var):
                missing_vars.append(var)

        if missing_vars:
            print(f"Error: The following variables are not set in {self.config_file}:")
            for var in missing_vars:
                print(f"  - {var}")
            sys.exit(1)

    def _uuid_to_dnid(self, cluster_uuid: str) -> str:
        """Convert UUID to DNID (simplified conversion)"""
        return cluster_uuid.replace('-', '_')

    def _construct_payload(self) -> Dict[str, Any]:
        """Construct the JSON payload for create_deployment (NO method field)

        Format:
        {
            "job": "create_deployment",
            "params": {
                "db_name": "...",
                ...
            }
        }
        """
        cluster_dnid = self._uuid_to_dnid(self.config["CLUSTER_UUID"])

        payload = {
            "job": "create_deployment",
            "params": {
                "db_name": self.config["DB_NAME"],
                "working_copy": self.config["WORKING_COPY"],
                "platform": self.config["PLATFORM"],
                "instance_type": self.config["INSTANCE_TYPE"],
                "num_nodes": self.config["NUM_NODES"],
                "size": self.config["SIZE"],
                "region": self.config["REGION"],
                "platform_reference": "",
                "account_uuid": self.config["ACCOUNT_UUID"],
                "db_uuid": self.config["DB_UUID"],
                "cluster_uuid": self.config["CLUSTER_UUID"],
                "cluster_dnid": cluster_dnid,
                "client_id": self.config["CLIENT_ID"],
                "max_multi_cluster_nodes": self.config["MAX_MULTI_CLUSTER_NODES"],
                "max_clusters": self.config["MAX_CLUSTERS"],
                "bucket_name" : self.config["BUCKET_NAME"]
            }
        }

        # Add optional network_cidr if provided
        if self.config.get("NETWORK_CIDR"):
            payload["params"]["network_cidr"] = self.config["NETWORK_CIDR"]

        return payload

    def _construct_job_exec_payload(self, job: str, job_params: Optional[Dict[str, Any]] = None, volatile: bool = True) -> Dict[str, Any]:
        """Construct the JSON payload for job_exec (NO method field)

        Format:
        {
            "job": "db_list",
            "params": {
                "params": {},
                "volatile": true
            }
        }
        """
        if job_params is None:
            job_params = {}

        payload = {
            "job": job,
            "params": {
                "params": job_params,
                "volatile": volatile
            }
        }

        return payload

    def _call_c4_api(self, payload: Dict[str, Any], attempt: int) -> Tuple[Dict[str, Any], bool]:
        """Make the API call to the C4 service using requests library"""
        url = f"{self.config['HOST_URL']}{self.config['ROUTE']}"
        username = self.config['USER']
        password = self.config['PASSWORD']

        print(f"Attempt {attempt}/{self.c4_retries}: Making request to {url}...")

        # Debug: Show request details
        debug_mode = self.config.get("DEBUG", "false").lower() == "true"
        if debug_mode:
            print("DEBUG - Request Details:")
            print(f"  URL: {url}")
            print(f"  User: {username}")
            print(f"  Payload: {json.dumps(payload, indent=2)}")
            print()

        try:
            start_time = time.time()

            response = requests.post(
                url,
                json=payload,
                auth=requests.auth.HTTPBasicAuth(username, password),
                verify=False,
                timeout=600
            )
            response.raise_for_status()
            json_result = response.json()

        except requests.exceptions.RequestException as e:
            error_msg = f"Failed to fetch data from the API: {str(e)}"
            print(f"Request Error: {error_msg}")
            return {"error": error_msg}, False

        except json.JSONDecodeError:
            error_msg = "Failed to parse the response as JSON."
            print(f"JSON Error: {error_msg}")
            return {"error": error_msg}, False

        end_time = time.time()

        print(f"Response (HTTP {response.status_code}):")
        print(json.dumps(json_result, indent=2))
        print(f"Request completed in {end_time - start_time:.2f} seconds")
        print()

        return json_result, True

    def create_deployment(self) -> Optional[Dict[str, Any]]:
        """Main deployment creation method with retry logic"""
        print("=== C4 Create Deployment Python Script ===")
        print(f"Using Basic Auth (user: {self.config['USER']})")
        print(f"Retry settings: {self.c4_retries} attempts, {self.c4_retry_sleep} seconds between retries")
        print()

        payload = self._construct_payload()

        print("Payload:")
        print(json.dumps(payload, indent=2))
        print()
        print("=" * 50)

        # Retry loop (matching the Python method logic with for...else pattern)
        for _ in range(self.c4_retries):
            result, success = self._call_c4_api(payload, _ + 1)

            if not success:
                print(f"c4 error on call create_deployment for database \"{self.config['CLUSTER_UUID']}\": {result}")
                time.sleep(self.c4_retry_sleep)
            else:
                # Success - break out of retry loop
                print("SUCCESS: Deployment request completed successfully!")
                return result
        else:
            # Loop completed without break - all retries failed
            print(f"ERROR: All {self.c4_retries} attempts failed. Deployment request unsuccessful.")
            return None

    def execute_request(self) -> Optional[Dict[str, Any]]:
        """Execute the configured request type (create_deployment or job_exec)"""
        request_type = self.config.get("REQUEST_TYPE", "job_exec")

        if request_type == "create_deployment":
            return self.create_deployment()
        elif request_type == "job_exec":
            job_name = self.config["JOB_NAME"]
            job_params = {}
            volatile = self.config.get("JOB_VOLATILE", "true").lower() == "true"

            # Parse job parameters if provided
            if self.config.get("JOB_PARAMS"):
                try:
                    job_params = json.loads(self.config["JOB_PARAMS"])
                except json.JSONDecodeError:
                    print(f"Warning: Invalid JOB_PARAMS JSON, using empty params")
                    job_params = {}

            return self.job_exec(job_name, job_params, volatile)
        else:
            print(f"Error: Unknown REQUEST_TYPE: {request_type}")
            return None

    def job_exec(self, job: str, job_params: Optional[Dict[str, Any]] = None, volatile: bool = True) -> Optional[Dict[str, Any]]:
        """Execute a generic job using job_exec method"""
        print(f"=== C4 Job Exec Python Script - Job: {job} ===")
        print(f"Using Basic Auth (user: {self.config['USER']})")
        print(f"Retry settings: {self.c4_retries} attempts, {self.c4_retry_sleep} seconds between retries")
        print()

        payload = self._construct_job_exec_payload(job, job_params, volatile)

        print("Payload:")
        print(json.dumps(payload, indent=2))
        print()
        print("=" * 50)

        # Retry loop with for...else pattern
        for _ in range(self.c4_retries):
            result, success = self._call_c4_api(payload, _ + 1)

            if not success:
                print(f"c4 error on job_exec for job \"{job}\": {result}")
                time.sleep(self.c4_retry_sleep)
            else:
                # Success - break out of retry loop
                print(f"SUCCESS: Job '{job}' completed successfully!")
                return result
        else:
            # Loop completed without break - all retries failed
            print(f"ERROR: All {self.c4_retries} attempts failed. Job '{job}' execution unsuccessful.")
            return None

def main():
    """Main entry point"""
    # Disable SSL warnings for self-signed certificates
    import urllib3
    urllib3.disable_warnings(urllib3.exceptions.InsecureRequestWarning)

    client = CreateDeploymentClient()
    client._validate_config()
    result = client.execute_request()

    if result is None:
        sys.exit(1)
    else:
        sys.exit(0)

if __name__ == "__main__":
    main()
